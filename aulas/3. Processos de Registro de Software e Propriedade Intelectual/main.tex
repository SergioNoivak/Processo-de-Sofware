% =========================================================
% SLIDE 1 — CONCEITO
% =========================================================
\begin{frame}{O que é Propriedade Intelectual?}

  \begin{center}
    \Large
    \textbf{Propriedade Intelectual (PI)}
  \end{center}

  \vspace{0.4cm}

  \begin{center}
    \normalsize
    É o conjunto de mecanismos jurídicos destinados a
    \textbf{proteger criações intelectuais},
    invenções e outros ativos resultantes da atividade humana.
  \end{center}

  \vspace{0.5cm}

  \begin{columns}[T]

    \column{0.31\textwidth}
    \begin{block}{O que pode ser protegido?}
      \small
      \begin{itemize}
        \item Invenções
        \item Marcas
        \item Obras artísticas
        \item Textos e músicas
        \item Programas de computador
        \item Conhecimento estratégico
      \end{itemize}
    \end{block}

    \column{0.31\textwidth}
    \begin{block}{O que a proteção permite?}
      \small
      \begin{itemize}
        \item Reconhecer a autoria
        \item Controlar determinados usos
        \item Impedir certas utilizações não autorizadas
        \item Explorar economicamente o ativo
      \end{itemize}
    \end{block}

    \column{0.31\textwidth}
    \begin{block}{Qual é a ideia central?}
      \small
      A PI busca criar um equilíbrio entre:

      \vspace{0.15cm}

      \centering
      \textbf{incentivar a criação}\\
      $\Downarrow$\\
      \textbf{proteger o criador}\\
      $\Downarrow$\\
      \textbf{beneficiar a sociedade}
    \end{block}

  \end{columns}

\end{frame}


% =========================================================
% SLIDE 2 — FINALIDADE
% =========================================================
\begin{frame}{Para que serve a Propriedade Intelectual?}

  \begin{center}
    \normalsize
    A proteção jurídica dos ativos intelectuais possui
    \textbf{funções econômicas, sociais e culturais}.
  \end{center}

  \vspace{0.35cm}

  \begin{columns}[T]

    \column{0.48\textwidth}

    \begin{block}{1. Incentivar a inovação}
      \small
      A possibilidade de obter proteção jurídica pode
      estimular pessoas e empresas a investir tempo,
      recursos e conhecimento no desenvolvimento de
      novas soluções.
    \end{block}

    \vspace{0.2cm}

    \begin{block}{2. Proteger o investimento}
      \small
      A criação de uma tecnologia, marca ou obra pode
      exigir investimentos significativos. A PI oferece
      instrumentos para proteger determinados resultados
      desse investimento.
    \end{block}

    \column{0.48\textwidth}

    \begin{block}{3. Permitir exploração econômica}
      \small
      Os direitos de PI podem ser utilizados para
      comercializar, licenciar, transferir ou explorar
      economicamente determinados ativos intelectuais.
    \end{block}

    \vspace{0.2cm}

    \begin{block}{4. Promover conhecimento e cultura}
      \small
      A proteção também faz parte de um sistema que
      busca estimular a produção científica, tecnológica,
      artística e cultural.
    \end{block}

  \end{columns}

  \vspace{0.35cm}

  \begin{alertblock}{Ponto importante}
    \small
    \textbf{Propriedade Intelectual não significa simplesmente
    ``ser dono de uma ideia''.} Cada modalidade possui
    requisitos, limites, duração e formas de proteção próprios.
  \end{alertblock}

\end{frame}

\begin{frame}{Propriedade Industrial}

  \begin{center}
    \small
    A \textbf{Propriedade Industrial} está especialmente relacionada
    à inovação, à atividade empresarial e à identificação de produtos
    e serviços no mercado.
  \end{center}

  \vspace{0.1cm}

  \begin{columns}[T]

    \column{0.48\textwidth}

    \begin{block}{Patentes}
      \small
      Protegem determinadas soluções técnicas que atendam
      aos requisitos legais.

      \vspace{0.1cm}

      \textbf{Exemplos:}
      novas máquinas, processos, produtos ou melhorias
      tecnológicas.
    \end{block}

    \begin{block}{Marcas}
      \small
      Identificam e distinguem produtos ou serviços
      de uma empresa em relação aos seus concorrentes.

      \vspace{0.1cm}

      \textbf{Exemplos:}
      nomes, sinais e elementos distintivos utilizados
      no mercado.
    \end{block}

    \column{0.48\textwidth}

    \begin{block}{Desenho Industrial}
      \small
      Relaciona-se aos aspectos ornamentais ou visuais
      de um produto.

      \vspace{0.1cm}

      \textbf{Exemplos:}
      forma, configuração ou composição visual.
    \end{block}

    \begin{block}{Indicação Geográfica}
      \small
      Relaciona determinado produto ou serviço a uma
      origem geográfica, observados os requisitos legais.

      \vspace{0.1cm}

      \textbf{Exemplo:}
      produtos reconhecidos por características
      associadas à sua região de origem.
    \end{block}

  \end{columns}

  \vspace{0.05cm}

  \begin{alertblock}{Pergunta para a turma}
    \small
    \textbf{Qual é a diferença entre proteger uma invenção,
    proteger uma marca e proteger a aparência de um produto?}
  \end{alertblock}

\end{frame}

\begin{frame}{Direito Autoral e outros mecanismos de proteção}

  \begin{columns}[T]

    \column{0.48\textwidth}

    \begin{block}{Direito Autoral}
      \small
      Protege obras intelectuais resultantes da criação
      humana, observados os requisitos e limites previstos
      na legislação.

      \vspace{0.15cm}

      \textbf{Exemplos:}
      \begin{itemize}
        \item Livros e artigos;
        \item Músicas;
        \item Fotografias e imagens;
        \item Filmes e outras obras audiovisuais;
        \item Obras artísticas;
        \item Programas de computador.
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}

    \begin{block}{Outros mecanismos}
      \small
      Nem todo ativo intelectual se encaixa exclusivamente
      em patente, marca ou direito autoral.

      \vspace{0.15cm}

      \begin{itemize}
        \item \textbf{Segredo empresarial:} informações estratégicas
        mantidas em sigilo;
        \item \textbf{Know-how:} conhecimentos e experiências
        relevantes para determinada atividade;
        \item \textbf{Informações confidenciais:} dados cujo acesso
        deve ser controlado;
        \item \textbf{Cultivares:} proteção específica relacionada
        a variedades vegetais.
      \end{itemize}
    \end{block}

  \end{columns}

  \vspace{0.25cm}

  \begin{block}{Uma mesma inovação pode envolver diferentes direitos}
    \small
    Um produto tecnológico, por exemplo, pode envolver
    \textbf{patente} para uma solução técnica,
    \textbf{marca} para sua identificação,
    \textbf{desenho industrial} para seu aspecto visual
    e \textbf{direito autoral} para determinados elementos de software.
  \end{block}

\end{frame}

\begin{frame}{Registro de Software vs. Patente}
  \begin{center}
    \large\color{mRed}\textbf{Diferença Fundamental!}\\
    \small Registro de software \textbf{não é a mesma coisa} que patente de software.
  \end{center}
  
  \vspace{0.2cm}
  
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Registro de Software}
        \small
        \textbf{Ramo:} Direito Autoral\\
        \textbf{O que protege:} A literalidade do código-fonte (texto escrito).\\
        \textbf{Requisitos:} Originalidade na escrita.\\
        \textbf{Processo:} Rápido, automático e declaratório no INPI.\\
        \textbf{Duração:} 50 anos a partir do ano seguinte à criação.
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Patente de Software}
        \small
        \textbf{Ramo:} Propriedade Industrial\\
        \textbf{O que protege:} A invenção/processo técnico (algoritmo associado a hardware).\\
        \textbf{Requisitos:} Novidade, atividade inventiva e aplicação industrial.\\
        \textbf{Processo:} Longo, com análise de mérito rigorosa pelo INPI.\\
        \textbf{Duração:} 20 anos a partir do depósito.
      \end{block}
    \end{column}
  \end{columns}
  
  \vspace{0.15cm}
  \begin{center}
    \scriptsize\color{structure.fg}\bfseries No Brasil, softwares em si (só o código) não são patenteáveis. Patenteia-se apenas invenções implementadas por computador.
  \end{center}
\end{frame}

\begin{frame}{O Mito da Patente de Algoritmo}
  \begin{center}
    \large\color{mRed}\textbf{Desconstruindo um erro comum:}\\
    \large\color{structure.fg}``Posso patentear meu código porque inventei um algoritmo.''
  \end{center}
  
  \vspace{0.2cm}
  
  \begin{block}{Por que essa afirmação está conceitualmente incorreta?}
    \small
    \begin{itemize}
      \item \textbf{Algoritmos puros são abstrações}: Métodos matemáticos, métodos de jogo, regras de negócio ou lógica computacional pura são descobertas ou representações lógicas, e a lei de propriedade industrial brasileira (LPI Art. 10) os exclui explicitamente de patenteabilidade.
      \item \textbf{Código-fonte puro é expressão}: A escrita do código é protegida por Direito Autoral (como uma obra literária), e não por patente.
      \item \textbf{Patente protege a aplicação técnica}: A patente é voltada para a solução de um problema técnico no mundo físico, não para ideias matemáticas puras.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Os Requisitos para Patenteabilidade}
  \begin{columns}[T]
    \begin{column}{0.65\textwidth}
      \begin{block}{Requisitos Gerais (LPI Art. 8)}
        \small
        \begin{itemize}
          \item \textbf{Novidade}: Não existir nada igual no mundo acessível ao público antes do depósito da patente (Estado da Técnica).
          \item \textbf{Atividade Inventiva}: A solução não pode ser óbvia ou decorrência comum para um especialista na área.
          \item \textbf{Aplicação Industrial}: A invenção deve poder ser fabricada ou utilizada industrialmente.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.33\textwidth}
      \begin{block}{Regra para IIC}
        \small
        Para softwares, exige-se o \textbf{efeito técnico adicional}: o software deve atuar controlando um processo físico ou modificando o estado de um dispositivo real, gerando utilidade palpável.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Hash Criptográfico e Integridade — 25 min}

\begin{frame}{Como identificar se um arquivo foi alterado?}
  \begin{center}
    \textbf{Pergunta de Transição:}\\[0.2cm]
    \large\color{structure.fg}
    ``Como podemos identificar se um arquivo ou código-fonte foi alterado?''
  \end{center}

  \vspace{0.3cm}

  \begin{block}{O que é um Hash Criptográfico?}
    \small
    Um \textbf{hash} é o resultado de uma função matemática determinística que transforma uma entrada de qualquer tamanho em uma sequência de caracteres de \textbf{tamanho fixo} (uma \textit{impressão digital} do arquivo).
  \end{block}

  \vspace{0.2cm}

  \begin{center}
    \begin{tikzpicture}[scale=0.85, every node/.style={transform shape}]
      \node[draw=structure.fg, fill=structure.fg!10, rounded corners=3pt, text width=2.8cm, align=center, font=\scriptsize] (file) at (-4,0) {
        \textbf{Arquivo}\\
        (Tamanho Arbitrário)
      };

      \node[softcore, minimum size=1.5cm, fill=structure.fg!30, font=\bfseries\tiny] (func) at (0,0) {
        Função\\Hash
      };

      \node[draw=mGreen!80!black, fill=mGreen!10, rounded corners=3pt, text width=3.2cm, align=center, font=\ttfamily\tiny] (hash) at (4,0) {
        \textbf{Hash (Digest)}\\
        3A7BD3E2360A3D...\\
        (Tamanho Fixo)
      };

      \draw[softarrow] (file) -- (func);
      \draw[softarrow] (func) -- (hash);
    \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{O Efeito Avalanche e a Integridade}
  \begin{center}
    \small\bfseries Mesmo a menor alteração no arquivo altera completamente seu Hash!
  \end{center}

  \vspace{0.2cm}

  \begin{center}
    \begin{tikzpicture}[scale=0.85, every node/.style={transform shape}]
      % Upper path (Original)

      \draw[softarrow] (algo1) -- (hash1);

  \vspace{0.2cm}

  \begin{alertblock}{Propriedade Fundamental}
    \small
    Não é possível prever o novo hash ao alterar o arquivo. Essa mudança drástica é conhecida como \textbf{Efeito Avalanche}, garantindo a verificação de integridade.
  \end{alertblock}
\end{frame}

\begin{frame}{Principais Algoritmos de Hash}
  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \begin{block}{MD5}
        \small
        \textbf{Saída:} 128 bits.\\
        \textbf{Status:} \color{mRed}\textbf{Obsoleto}.\\
        \vspace{0.1cm}
        \scriptsize Vulnerável a colisões conhecidas. Não deve ser usado para segurança.
      \end{block}
    \end{column}

    \begin{column}{0.31\textwidth}
      \begin{block}{SHA-1}
        \small
        \textbf{Saída:} 160 bits.\\
        \textbf{Status:} \color{orange!80!black}\textbf{Descontinuado}.\\
        \vspace{0.1cm}
        \scriptsize Desaconselhado para novos usos criptográficos devido a ataque de colisão.
      \end{block}
    \end{column}

    \begin{column}{0.34\textwidth}
      \begin{block}{SHA-256}
        \small
        \textbf{Saída:} 256 bits.\\
        \textbf{Status:} \color{mGreen!80!black}\textbf{Recomendado}.\\
        \vspace{0.1cm}
        \scriptsize Padrão atual seguro para verificação de integridade, assinaturas e Git.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4cm}

  \begin{center}
    \small\color{structure.fg}\bfseries O que é uma colisão? É quando duas entradas diferentes geram exatamente o mesmo hash!
  \end{center}
\end{frame}

\section{Prática: Gerando um Hash — 20 min}

\begin{frame}[fragile]{Prática: Gerando Hashes no Terminal}
  \begin{block}{Comandos por Sistema Operacional}
    \scriptsize
    \textbf{Windows (PowerShell):}
    \begin{verbatim}
Get-FileHash .\software.zip -Algorithm SHA256
    \end{verbatim}

    \textbf{Linux:}
    \begin{verbatim}
sha256sum software.zip
    \end{verbatim}

    \textbf{macOS:}
    \begin{verbatim}
shasum -a 256 software.zip
    \end{verbatim}
  \end{block}

  \begin{block}{Experimento Prático Passo a Passo}
    \small
    \begin{enumerate}
      \item Crie o arquivo \texttt{codigo.txt} contendo uma frase.
      \item Gere o SHA-256 do arquivo e anote o resultado.
      \item Altere uma única letra ou espaço no arquivo.
      \item Gere o SHA-256 novamente e compare os valores.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Desmistificando o Hash (Debate com a Turma)}
  \begin{center}
    \large\color{structure.fg}\textbf{Entendendo a Natureza do Hash}
  \end{center}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pergunta 1}
        \small
        \textbf{“O hash permite recuperar o conteúdo original do arquivo?”}
      \end{alertblock}
      \vspace{0.1cm}
      \begin{block}{Resposta: NÃO}
        \small
        O hash é uma \textbf{função unidirecional (one-way)}. Não há algoritmo inverso para reconstruir os dados originais a partir da sequência de hash.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pergunta 2}
        \small
        \textbf{“Hash é a mesma coisa que Criptografia?”}
      \end{alertblock}
      \vspace{0.1cm}
      \begin{block}{Resposta: NÃO}
        \small
        Criptografia visa ocultar dados de forma \textbf{reversível} (com chave). Hash é um \textbf{resumo/digest} irreversível para verificação de integridade.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ferramentas e Mecanismos Complementares}
  \begin{center}
    \small\bfseries Como transformar o Hash em evidência jurídica robusta?
  \end{center}

  \vspace{0.1cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Git \& Commits}
        \small
        Histórico encadeado de hashes de commits que registram a evolução temporal do projeto.
      \end{block}

      \begin{block}{Timestamps / Carimbo do Tempo}
        \small
        Registrar o hash em uma Autoridade Certificadora de Tempo (TSA) ou blockchain comprova existência na data.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Assinatura Digital \& Certificados}
        \small
        Vincular o hash à chave privada do autor (ex: ICP-Brasil) garante a \textbf{autoria} do documento.
      \end{block}

      \begin{block}{Registro de Software (INPI)}
        \small
        Fornece a presunção \textbf{legal} oficial de autoria e anterioridade perante o Estado.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
