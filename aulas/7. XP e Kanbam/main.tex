\section{Atividade 1: Planejamento (XP)}

% Slide 5: O Jogo do Planejamento (com Diagrama Cliente x Equipe)

  \begin{block}{O Início: "Ouvir"}
    A atividade de planejamento (conhecida como \textbf{O Jogo do Planejamento}) começa com o ato de \textbf{ouvir} o cliente.

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Histórias de Usuário}
        \begin{itemize}
          \item Escritas em fichas pelo cliente.
          \item Criadas a qualquer momento do projeto.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Atribuição de Valor vs. Custo}
        \begin{itemize}
          \item \textbf{Cliente:} Define o Valor de Negócio.
          \item \textbf{Devs:} Estimam o Custo em semanas.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 6: Agrupamento e Ordenação das Histórias
\begin{frame}{Planejamento: Ordenação e Compromisso de Versão}
  \begin{block}{Compromisso de Versão (\textit{Release Commitment})}
    Clientes e desenvolvedores trabalham em conjunto para agrupar as histórias que comporão o \textbf{próximo incremento de software} (próxima versão), pactuando datas de entrega e escopo.
  \end{block}

  \vspace{0.3cm}
  
  \textbf{Formas de Ordenação das Histórias para Desenvolvimento:}
  \begin{enumerate}
    \item \textbf{Implementação Imediata:} Todas as histórias selecionadas serão implementadas imediatamente em um curto prazo (poucas semanas).
    \item \textbf{Maior Valor Primeiro:} As histórias com maior valor de negócio são movidas para o topo do cronograma e desenvolvidas primeiro.
    \item \textbf{Maior Risco Primeiro:} As histórias que apresentam maior risco técnico ou operacional sobem no cronograma para mitigação antecipada.
  \end{enumerate}
\end{frame}

% Slide 7: Velocidade do Projeto
\begin{frame}{Planejamento: Velocidade do Projeto (\textit{Project Velocity})}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{O que é a Velocidade do Projeto?}
        É o \textbf{número de histórias de clientes efetivamente implementadas} durante a entrega da primeira versão/incremento do software.
      \end{block}
      
      \vspace{0.2cm}
      
      \begin{exampleblock}{Utilidade Prática}
        \begin{itemize}
          \item Utilizada como métrica empírica real da equipe.
          \item Ajuda a estimar datas de entrega e cronogramas das \textbf{versões subsequentes}.
          \item Substitui estimativas teóricas por dados reais de produtividade.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    
    \begin{column}{0.41\textwidth}
      \begin{alertblock}{Adaptação Contínua}
        A velocidade não é rígida. A equipe XP \textbf{modifica e adapta seus planos continuamente} à medida que a velocidade do projeto oscila ou se consolida ao longo das iterações.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Atividade 2: Projeto (XP)}

% Slide 8: O Princípio KISS no Projeto XP
\begin{frame}{Atividade 2: Projeto --- O Princípio KISS}
  \begin{block}{Princípio KISS (\textit{Keep It Simple, Stupid!})}
    O projeto XP segue rigorosamente o princípio da simplicidade: \textbf{"Não complique!"}.
  \end{block}

  \vspace{0.3cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{O que é Desestimulado?}
        \begin{itemize}
          \item \textbf{Projeto Especulativo:} Projetar funcionalidades extras prevendo que "serão necessárias no futuro".
          \item Complexidade arquitetural antecipada e desnecessária.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Foco na Necessidade Atual}
        \begin{itemize}
          \item O projeto deve atender \textbf{apenas e exatamente} às histórias do incremento atual.
          \item Projetos simples são mais fáceis de modificar quando novas necessidades surgirem.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 10: Refatoração Contínua
\begin{frame}{Projeto: Refatoração Contínua (\textit{Refactoring})}
  \begin{block}{Projeto Dinâmico e Incremental}
    Na XP, a elaboração do projeto \textbf{não é uma fase estática}: ocorre tanto \textbf{antes} quanto \textbf{depois} de se ter iniciado a codificação. A própria atividade de desenvolvimento guia o aprimoramento do projeto.
  \end{block}

  \vspace{0.3cm}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{exampleblock}{O Conceito de Refatoração (\textit{Fowler, 2000})}
        \textit{"Modificar e otimizar a estrutura interna do código sem alterar o seu comportamento externo."}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.41\textwidth}
      \begin{alertblock}{Projetar Continuamente}
        Significa que o projeto é aperfeiçoado de forma ininterrupta enquanto o sistema está sendo construído.
      \end{alertblock}
    \end{column}

    Exemplo de teste unitário limpo exercitando o cálculo de desconto para garantir regressão automatizada.
  \end{block}

  \begin{lstlisting}[style=javastyle]
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;
import static org.junit.jupiter.api.Assertions.*;

public class CalculadoraDescontoTest {
    private CalculadoraDesconto calculadora;

    @BeforeEach
    void setUp() {
        calculadora = new CalculadoraDesconto(); // Arrange comum
    }

    @Test
    @DisplayName("Deve aplicar 10% de desconto para compras acima de R$ 500,00")
    void deveAplicarDescontoParaComprasAcimaDoLimite() {
        // 1. Arrange (Preparacao dos dados)
        double valorPedido = 600.00;

        // 2. Act (Execucao do metodo testado)
        double valorFinal = calculadora.calcular(valorPedido);

        // 3. Assert (Verificacao do resultado)
        assertEquals(540.00, valorFinal, 0.001, "Desconto de 10% incorreto");
    }
}
  \end{lstlisting}
\end{frame}

    @DisplayName("Deve lancar IllegalArgumentException para valor de pedido negativo")

        );

}

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ProcessadorDePagamentoTest {

    @Mock
    private GatewayPagamentoPix gatewayMock; // Dependencia externa simulada

    @InjectMocks
    private ProcessadorDePagamento processador; // Objeto sob teste

    @Test
    void deveAprovarTransacaoQuandoGatewayRetornarSucesso() {
        // Arrange: Configura expectativa no Mock
        when(gatewayMock.autorizar("TX123", 150.00)).thenReturn(true);

        // Act: Executa a regra de negocio
        boolean resultado = processador.processar("TX123", 150.00);

        // Assert & Verify: Valida resultado e chamada obrigatoria
        assertTrue(resultado);
        verify(gatewayMock, times(1)).autorizar("TX123", 150.00);
    }
}
  \end{lstlisting}
\end{frame}

\section{Panorama do Método Kanban}

% Slide 20: Origem e Conceito do Kanban
\begin{frame}{Panorama do Método Kanban}
  \begin{block}{O que é o Método Kanban?}
    O método Kanban (\textit{Anderson, 2016}) é uma \textbf{metodologia enxuta (\textit{Lean})} que descreve abordagens práticas para gerenciar e melhorar qualquer processo ou fluxo de trabalho.
  \end{block}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Origem e Adaptação}
        \begin{itemize}
          \item Nasceu na \textbf{Toyota} como um conjunto de práticas de engenharia industrial (sistema puxado).
          \item Adaptado para o desenvolvimento de software por \textbf{David Anderson}.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pilares de Gestão}
        \begin{description}[Serviços]
          \item[\textbf{Gestão de Alterações:}] Integração sistemática de solicitações ao software.
          \item[\textbf{Entrega de Serviços:}] Foco total no entendimento das necessidades do cliente.
          \item[\textbf{Auto-organização:}] Membros têm liberdade para organizar seu trabalho.
        \end{description}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 21: As 6 Práticas Fundamentais do Kanban
\begin{frame}{As 6 Práticas Fundamentais do Kanban}
  \begin{block}{Práticas Fundamentais (David Anderson)}
    O método Kanban fundamenta-se em \textbf{seis práticas chave} para a otimização contínua de processos:
  \end{block}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{enumerate}
        \item \textbf{Visualizar o fluxo de trabalho} (uso do Quadro Kanban).
        \item \textbf{Limitar o estoque em processo} (WIP --- \textit{Work In Progress}).
        \item \textbf{Gerenciar o fluxo de trabalho} (redução de desperdícios e gargalos).
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{enumerate}
        \setcounter{enumi}{3}
        \item \textbf{Explicitar as políticas do processo} (critérios claros e visíveis).
        \item \textbf{Enfocar a melhoria contínua} (ciclos de feedback orientados por dados).
        \item \textbf{Alterar o processo colaborativamente} (engajamento de todo o time).
      \end{enumerate}
    \end{column}
  \end{columns}

\section{Práticas Kanban em Detalhe}

% Slide 22: Prática 1 - Visualizar o Fluxo de Trabalho
\begin{frame}{Prática 1: Visualizar o Fluxo de Trabalho}
  \begin{block}{O Quadro Kanban (\textit{Kanban Board})}
    O fluxo de trabalho é representado graficamente em um \textbf{quadro dividido em colunas}, que representam as diferentes etapas de desenvolvimento de cada item.
  \end{block}

  \vspace{0.3cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Elementos de Trabalho (Cartões)}
        \begin{itemize}
          \item Cartões em notas adesivas (\textit{post-its}) representam \textbf{histórias de usuário} ou \textbf{defeitos recém-descobertos}.
          \item Os cartões contêm informações essenciais do item de trabalho.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Movimentação Puxada (\textit{Pull System})}
        \begin{itemize}
          \item A equipe movimenta os cartões da coluna \textbf{"Por Fazer"} $\rightarrow$ \textbf{"Fazendo"} $\rightarrow$ \textbf{"Feito"}.
          \item Dá visibilidade total do status de cada funcionalidade em tempo real.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

        \begin{itemize}

          \item Analisa os pontos em que o trabalho sofre interrupções ou acúmulo (\textbf{gargalos}).

        \begin{itemize}
          \item As regras do processo devem ser claras e visíveis para todos.
          \item Registra os motivos de seleção de cada item de trabalho.
          \item Define explicitamente os critérios para um item ser considerado \textbf{"Feito"} (\textit{Definition of Done - DoD}).
        \end{itemize}
      \end{exampleblock}
    \end{column}

\begin{frame}{Reuniões de Equipe e Inicialização em Projetos Existentes}
  \begin{block}{Alinhamento Contínuo e Semelhança com o Scrum}
    As reuniões de equipe para o Kanban são \textbf{semelhantes àquelas realizadas no Scrum}, mantendo a equipe focada no fluxo de trabalho e na resolução rápida de impedimentos.
  \end{block}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Introdução em Projetos Existentes}
        \begin{itemize}
          \item Nem todos os itens de trabalho iniciam na coluna do \textit{Backlog}.
          \item Mapeia-se o estado real e corrente de cada tarefa em andamento.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{As 3 Perguntas Fundamentais}
        Para posicionar os cartões corretamente no processo, os desenvolvedores perguntam:
        \begin{enumerate}
          \item \textbf{Onde estão agora?}
          \item \textbf{De onde vieram?}
          \item \textbf{Aonde vão?}
        \end{enumerate}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 28: A Reunião Diária (Standup Meeting) no Kanban
\begin{frame}{A Reunião Diária (\textit{Standup Meeting}) e "Caminhar pelo Quadro"}
  \begin{block}{A Prática de "Caminhar pelo Quadro" (\textit{Walking the Board})}
    A base para a reunião diária em pé do Kanban é \textbf{"caminhar pelo quadro"}. A liderança/facilitador da reunião muda diariamente entre os membros do time.
  \end{block}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Mapeamento e Avanço de Itens}
        \begin{itemize}
          \item Identificação e adição de itens ausentes que estão sendo trabalhados.
          \item \textbf{Objetivo Principal:} Avançar o máximo possível de itens para a coluna \textbf{"Feito"}.
          \item Prioridade: Deslocar primeiro os itens de \textbf{alto valor de negócio}.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Identificação de Obstáculos}
        \begin{itemize}
          \item Observação ativa do fluxo do quadro.
          \item Análise da \textbf{carga de trabalho} e dos \textbf{riscos} para remover impedimentos rapidamente.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Panorama e Origem do DevOps}

% Slide 30: Origem e Conceito de DevOps
\begin{frame}{Panorama Geral do DevOps}
  \begin{block}{O que é o DevOps?}
    Criado por \textbf{Patrick DeBois} (\textit{Kim et al., 2016}), o DevOps une as áreas de \textbf{Desenvolvimento} (\textit{Development}) e \textbf{Operações} (\textit{Operations}) em um ciclo integrado.
  \end{block}

  \vspace{0.2cm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Extensão dos Princípios Ágeis}
        \begin{itemize}
          \item Aplica os princípios de desenvolvimento \textbf{ágil e enxuto (\textit{Lean})} a toda a cadeia logística de software.
          \item Substitui os tradicionais "silos" entre Dev e Ops por uma cultura colaborativa e contínua.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Ciclos Contínuos}
        \begin{itemize}
          \item O processo envolve diversas etapas interconectadas que formam \textbf{ciclos contínuos}.
          \item Permanece ativo até que o produto desejado exista e opere em produção com qualidade.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{As 5 Práticas Contínuas do DevOps}

% Slide 32: Desenvolvimento e Teste Contínuos
\begin{frame}{DevOps: Desenvolvimento e Teste Contínuos}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Desenvolvimento Contínuo}
        \begin{itemize}
          \item Produtos de software divididos e construídos em \textbf{múltiplos \textit{sprints}}.
          \item Incrementos frequentes entregues diretamente aos membros de Garantia da Qualidade (QA) da equipe.
          \item Foco no alinhamento contínuo das novas funcionalidades.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Teste Contínuo}
        \begin{itemize}
          \item Uso intensivo de \textbf{ferramentas de teste automatizadas}.
          \item Teste simultâneo de múltiplos incrementos de código.
          \item Garantia de zero defeitos antes que o código siga para a etapa de integração.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
