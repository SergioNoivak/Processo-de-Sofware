\section{O Manifesto Ágil}

\subsection{Origem e Propósito}
\begin{frame}{Origem do Manifesto Ágil (2001)}
  \begin{block}{O Marco Histórico}
    Em \textbf{2001}, um grupo composto por 17 desenvolvedores de software, autores e consultores de renome [Beck et al., 2001] assinou o \textit{“Manifesto para o Desenvolvimento Ágil de Software”}.
  \end{block}

  \vspace{0.3cm}
  \begin{alertblock}{Propósito}
    Propor uma alternativa inovadora aos modelos de processo prescritivos e rígidos, sanando fraquezas da engenharia de software convencional.
  \end{alertblock}

  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Contexto:} Ambientes acelerados demandavam maior flexibilidade.
    \item \textbf{Foco:} Valorizar as pessoas, a entrega contínua de valor e a adaptação a mudanças.
  \end{itemize}
\end{frame}

\section{O que é Agilidade? (Seção 3.1)}

\begin{frame}{O que é Agilidade no Contexto de Software?}
  \begin{block}{O Condutor da Agilidade (Jacobson)}
    Ivar Jacobson [Jac02a] argumenta que a \textbf{difusão da mudança} é o principal condutor para a agilidade.
    \begin{itemize}
      \item Engenheiros de software devem ser rápidos para assimilar rápidas mudanças no ambiente.
    \end{itemize}
  \end{block}

  \vspace{0.2cm}
  \begin{alertblock}{Além da Simples Resposta à Mudança}
    Agilidade abrange uma \textbf{filosofia e cultura} de trabalho estruturada em diversos pilares, que serão detalhados a seguir: comunicação, foco em entrega, redução de artefatos intermediários e planejamento flexível.
  \end{alertblock}
\end{frame}

\begin{frame}{Foco no Software e Redução de Artefatos}
  \begin{block}{Software Operacional Acima de Tudo}
    A ênfase primária é a \textbf{entrega rápida e contínua} de software que funciona, em vez de documentação extensa.
  \end{block}

  \vspace{0.4cm}
  \centering
  \begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]
    % Traditional
    \node[draw, fill=red!10, thick, text width=3.5cm, align=center, rounded corners] (trad) at (0, 0) {
      \textbf{Tradicional}\\
      \footnotesize Excesso de modelos,\\diagramas e papéis
    };
    
    % Agile
    \node[draw, fill=green!10, thick, text width=3.5cm, align=center, rounded corners] (agile) at (6, 0) {
      \textbf{Ágil}\\
      \footnotesize Apenas artefatos que\\agregam valor imediato
    };

    % Arrow
    \draw[-{Latex[length=3mm]}, ultra thick] (trad) -- (agile) node[midway, above, font=\scriptsize, text width=2cm, align=center] {Redução de\\Desperdício};
  \end{tikzpicture}

  \vspace{0.4cm}
  \begin{itemize}
    \item \textbf{Exemplo:} Em vez de criar um documento de requisitos de 100 páginas, a equipe ágil mantém um conjunto priorizado de \textit{Histórias de Usuário}.

\section{Agilidade e o Custo das Mudanças (Seção 3.2)}

\begin{frame}{Agilidade e o Custo das Mudanças}
  \begin{block}{O Senso Comum na Engenharia de Software}
    Décadas de experiência demonstram que os \textbf{custos de mudanças aumentam de forma não linear} (exponencial) conforme o projeto avança.
  \end{block}

  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Mudança no Início (Requisitos)}
        \begin{itemize}
          \item \textbf{Ações:} Editar especificação, alterar detalhamento de uso ou ampliar lista de funções.
          \item \textbf{Custo:} Mínimo.
          \item \textbf{Impacto:} Praticamente nulo no cronograma final.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Mudança Tardia (Validação)}
        \begin{itemize}
          \item \textbf{Ações:} Alterar arquitetura, criar 3 novos componentes, modificar 5 existentes, novos testes.
          \item \textbf{Custo:} Cresce rapidamente.
          \item \textbf{Impacto:} Esforço massivo para evitar efeitos colaterais.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

  Se uma grande mudança funcional for solicitada durante os \textbf{testes de validação}:

\section{O Processo Ágil}

\begin{frame}{O que é Processo Ágil? -- Preceitos}
  Um processo ágil lida com a imprevisibilidade lidando com três preceitos fundamentais [Fow02]:
  
  \vspace{0.2cm}
  \begin{enumerate}
    \item \textbf{Dificuldade de Previsão:} É difícil prever quais requisitos persistirão e como as prioridades do cliente sofrerão alterações.
    \item \textbf{Projeto e Construção Intercalados:} Modelos de projeto precisam ser testados durante a construção. É difícil saber quanto de projeto é necessário antes da implementação.
    \item \textbf{Planejamento Limitado:} Análise, projeto, construção e testes não são tão previsíveis do ponto de vista de planejamento.
  \end{enumerate}
\end{frame}

\begin{frame}{Administrando a Imprevisibilidade}
  \begin{block}{Adaptabilidade do Processo}
    Para administrar a imprevisibilidade, o processo deve ser \textbf{adaptável} para alterar rapidamente o projeto e as condições técnicas.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Adaptação Incremental e Feedback}
    \begin{itemize}
      \item A equipe precisa do \textbf{feedback do cliente} para fazer adaptações apropriadas.
      \item O \textbf{protótipo operacional} (ou incremento executável) é o catalisador eficaz para esse feedback.
      \item Entregas curtas capacitam o cliente a avaliar o incremento e influenciar o processo regularmente.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Os 12 Princípios}
\begin{frame}{Princípios da Agilidade (Agile Alliance)}
  A satisfação do cliente é alcançada com a \textbf{oferta de valor e entregas frequentes}.
  
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Equipe e Ambiente:} Indivíduos motivados, comunicação presencial e auto-organização.
    \item \textbf{Foco Técnico:} Excelência técnica e bom projeto. 
    \item \textbf{Simplicidade:} A arte de maximizar o volume de trabalho \textit{não realizado}.
    \item \textbf{Software Operacional:} O objetivo principal, desenvolvido em um ritmo sustentável a longo prazo.
    \item \textbf{Introspecção e Melhoria:} A equipe busca continuamente melhorar sua forma de trabalho.
  \end{itemize}
\end{frame}

\begin{frame}{A Política do Desenvolvimento Ágil}
  O debate metodológico entre agilistas e tradicionais não deve virar uma "guerra santa".
  
  \vspace{0.2cm}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Visão Ágil Extremista}
        Os metodologistas tradicionais focam em documentação excessiva em vez de sistemas úteis e operacionais.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Visão Tradicional Extremista}
        Os ágeis são "hackers" que terão surpresas ao lidar com softwares de porte empresarial.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{block}{O Caminho do Meio}
    Software funcional é importante, mas também precisa de confiabilidade, usabilidade e facilidade de manutenção. Deve-se \textbf{mesclar o melhor das duas escolas}.
  \end{block}
\end{frame}


\section{Scrum}

\subsection{Fundamentos e Artefatos}
\begin{frame}{O Método Scrum}
  \begin{block}{O que é?}
    \textbf{Scrum} (nome derivado do \textit{rugby}) é um método ágil concebido por Jeff Sutherland (início dos anos 1990) e expandido por Schwaber e Beedle.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Princípios e Atividades}
    Coerente com o manifesto ágil, o Scrum incorpora atividades de \textbf{requisitos, análise, projeto, evolução e entrega}.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Sprints}
    O trabalho ocorre em janelas de tempo limitadas (\textit{time-boxes}) chamadas \textbf{sprints}. O número de sprints varia conforme o projeto, e o processo é adaptado pela equipe em tempo real.
  \end{block}
\end{frame}

\begin{frame}{Equipe Scrum e Artefatos (1/2)}
  \begin{block}{A Equipe Scrum}
    É interdisciplinar, auto-organizada e pequena (3 a 6 desenvolvedores), composta por:
    \begin{itemize}
      \item \textbf{Product Owner:} Cuida dos interesses do cliente/negócio.
      \item \textbf{Scrum Master:} Facilitador que remove obstáculos.
      \item \textbf{Equipe de Desenvolvimento.}
    \end{itemize}
  \end{block}

  \vspace{0.2cm}
  \begin{block}{O Backlog do Produto}
    Uma \textbf{lista priorizada} de requisitos que agregam valor ao negócio. É dinâmico e ordenado pelo \textit{Product Owner}.
  \end{block}
\end{frame}

\begin{frame}{Equipe Scrum e Artefatos (2/2)}
  \begin{block}{O Backlog do Sprint}
    Subconjunto do backlog do produto selecionado para ser completado no \textit{sprint} atual (geralmente 2 a 4 semanas). Não se adiciona novas características durante o sprint.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{O Incremento de Código}
    União de itens completados nos sprints anteriores com os itens do sprint atual. A equipe decide como completá-lo e quando está pronto.
  \end{block}
  
  \vspace{0.2cm}
  \begin{block}{O Papel do Scrum Master}
    Comanda a reunião diária de 15 minutos, remove obstáculos, ajuda a equipe e o \textit{product owner} a gerenciar o backlog.
  \end{block}
\end{frame}

\subsection{Cerimônias Scrum (Eventos)}
\begin{frame}{Reunião de Planejamento do Sprint}
  \begin{block}{Preparação e Priorização}
    A equipe e o \textit{product owner} desenvolvem e ordenam os itens do backlog pela importância para o negócio e pela complexidade técnica.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Definindo a Meta do Sprint}
    Antes do sprint, o \textit{product owner} enuncia a \textbf{meta de desenvolvimento}. O Scrum Master e a equipe selecionam os itens para o \textit{backlog do sprint}.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Comprometimento da Equipe}
    A equipe determina o que pode entregar no incremento dentro da janela de tempo e define os papéis necessários para a execução do trabalho.
  \end{block}
\end{frame}

\begin{frame}{Reunião Diária do Scrum (Cont.)}
  \begin{block}{Objetivos}
    Revelar problemas cedo. O Scrum Master deve focar em \textbf{eliminar os obstáculos} antes do próximo encontro. Não é uma reunião para solucionar problemas (isso é feito à parte). Promove a "socialização do conhecimento".
  \end{block}
\end{frame}

\begin{frame}{Reunião de Revisão do Sprint}
  \begin{block}{Quando e Quem?}
    Ocorre ao final do sprint (ex: 4 horas de reunião para sprint de 4 semanas). Participam: Scrum Master, Equipe de Desenvolvimento, Product Owner e envolvidos.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{A Demonstração (Demo)}
    A atividade principal é uma \textbf{demonstração do incremento de software} completado (funções entregues na janela de tempo, não necessariamente tudo o que foi planejado).
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Aceitação e Feedback}
    O \textit{Product Owner} pode ou não aceitar o incremento. Feedbacks são recolhidos para a próxima rodada de planejamento, onde novas características podem ser adicionadas ou eliminadas do backlog.
  \end{block}
\end{frame}

\begin{frame}{Retrospectiva do Sprint}
  \begin{block}{Objetivo e Duração}
    Realizada após a Revisão e antes do próximo Planejamento (ex: 3 horas para sprint de 4 semanas). 
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Debate da Equipe}
    O Scrum Master encoraja a equipe a debater e responder:
    \begin{itemize}
      \item O que deu certo no sprint?
      \item O que poderia melhorar?
      \item Com o que a equipe se compromete a melhorar no próximo sprint?
    \end{itemize}
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Melhoria Contínua}
    A equipe busca melhorar suas práticas, adaptar a definição de "pronto" e se preparar efetivamente para o próximo incremento.
  \end{block}
\end{frame}

\subsection{Aprofundamento: Casos de Teste}
\begin{frame}{Casos de Teste (Test Cases)}
  \begin{block}{O que é um Caso de Teste?}
    É um conjunto de \textbf{condições, passos e variáveis} sob as quais um Analista de Qualidade (QA) ou testador determinará se um requisito de software (como uma História de Usuário) foi implementado corretamente e funciona conforme esperado.
  \end{block}
  
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{No Azure DevOps:} São representados como \textit{Work Items} específicos (geralmente organizados dentro de um "Test Plan").
    \item \textbf{Composição Básica:} Possuem \textit{Pré-condições}, \textit{Ações (Passos a executar)}, e o \textit{Resultado Esperado} para cada passo.
    \item \textbf{Rastreabilidade:} Ficam vinculados diretamente à História de Usuário que estão validando. Se a User Story muda, o Caso de Teste deve ser revisado e atualizado.
  \end{itemize}
\end{frame}

\begin{frame}{Execução de um Caso de Teste}
  A execução de um Caso de Teste no ciclo ágil garante que novas funcionalidades entreguem o valor esperado e não quebrem o restante do sistema.

  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Manual vs Automatizado:} Podem ser executados manualmente por um humano (clicando e validando visualmente) ou via scripts automatizados (ex: Cypress, Selenium, JUnit).
    \item \textbf{Passo a Passo (Test Runner):} Em ferramentas como Azure DevOps, o QA usa um painel chamado \textit{Test Runner} que exibe o passo a passo na tela. Para cada passo, ele marca como \textcolor{green!60!black}{\textbf{Passou (Passed)}} ou \textcolor{red!80!black}{\textbf{Falhou (Failed)}}.
    \item \textbf{Evidências e Bugs:} Se um passo falha, o testador anexa evidências (prints de tela, logs de rede) e abre um \textit{Bug} diretamente vinculado àquele passo que falhou, bloqueando a entrega final da User Story até a correção.
  \end{itemize}
\end{frame}

\section{Conceito e Filosofia Ágil}

\begin{frame}{O que é a Engenharia de Software Ágil?}
  A Engenharia de Software Ágil combina uma \textbf{filosofia} com um conjunto de \textbf{princípios de desenvolvimento}.

  \vspace{0.3cm}
  \begin{block}{A Filosofia Ágil}
    \begin{itemize}
      \item Satisfação do cliente por meio da \textbf{entrega incremental antecipada};
      \item Equipes de projeto \textbf{pequenas e altamente motivadas};
      \item Utilização de \textbf{métodos informais} e artefatos mínimos;
      \item Acima de tudo: \textbf{simplicidade} no desenvolvimento geral.
    \end{itemize}
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Princípios de Desenvolvimento}
    \begin{itemize}
      \item Priorizam a \textbf{entrega} mais do que a análise e o projeto (embora essas atividades não sejam desencorajadas).
    \end{itemize}
  \end{block}
\end{frame}


\section{Equipe e o Fator Humano}

\subsection{O Papel da Equipe}
\begin{frame}{Quem Realiza? A Equipe Ágil}
  Engenheiros de software, gerentes, clientes e usuários trabalham conjuntamente em uma \textbf{equipe ágil}.

  \vspace{0.3cm}
  \begin{block}{Características Fundamentais}
    \begin{itemize}
      \item \textbf{Auto-organização:} A equipe controla seu próprio destino e toma suas decisões.
      \item \textbf{Comunicação Acelerada:} Elimina barreiras e aproxima todos os participantes.
      \item \textbf{Colaboração Contínua:} Todos os envolvidos atuam integrados a serviço do produto.
    \end{itemize}
  \end{block}

\subsection{O Fator Humano (Cockburn)}
\begin{frame}{O Fator Humano nos Processos (Cockburn)}
  Alistair Cockburn [Cockburn, 2002] destaca uma falha essencial dos modelos prescritivos tradicionais:

  \vspace{0.2cm}
  \begin{alertblock}{Desenvolvedores Não são Robôs}
    Engenheiros de software apresentam grande variação individual em:
    \begin{itemize}
      \item Estilos de trabalho e níveis de habilidade;
      \item Criatividade, organização, consistência e espontaneidade;
      \item Habilidade de comunicação escrita.
    \end{itemize}
  \end{alertblock}

  \vspace{0.2cm}
  \begin{block}{Requisito para os Modelos de Processo}
    Para funcionarem na prática, os processos devem:
    \begin{enumerate}
      \item Fornecer um mecanismo realista que estimule a \textbf{disciplina} necessária; OU
      \item Apresentar \textbf{tolerância} com as pessoas que realizam o trabalho.
    \end{enumerate}
  \end{block}
\end{frame}

\section{Aplicabilidade e Limitações}

\begin{frame}{Aplicabilidade e Limitações dos Métodos Ágeis}
  \begin{block}{Objetivo dos Métodos Ágeis}
    Sanar fraquezas, supostas e reais, da engenharia de software convencional, entregando sistemas corretos rapidamente.
  \end{block}

  \vspace{0.2cm}
  \begin{alertblock}{Limitações Importantes}
    O desenvolvimento ágil oferece grandes benefícios, porém:
    \begin{itemize}
      \item \textbf{Não é indicado para todos} os projetos, produtos, pessoas e situações.
    \end{itemize}
  \end{alertblock}

  \vspace{0.2cm}
  \begin{block}{Visão Prática}
    Não é a antítese da prática confiável — pode ser aplicado como uma \textbf{filosofia geral} para todos os trabalhos de software.
  \end{block}
\end{frame}
\begin{frame}{Quando NÃO utilizar Métodos Ágeis?}
  \begin{block}{Critérios de Inviabilidade}
    O desenvolvimento ágil pode não ser a melhor abordagem quando:
  \end{block}

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Requisitos Estritamente Fixos:} O escopo, o orçamento e os prazos são inflexíveis (ex: determinados contratos governamentais).
    \item \textbf{Sistemas Críticos:} Projetos com alto risco de vida ou regulação severa (ex: softwares aeroespaciais, equipamentos médicos), que exigem documentação rigorosa e design prévio detalhado (BDUF - \textit{Big Design Up Front}).
    \item \textbf{Cliente Indisponível:} O cliente ou representante (Product Owner) não tem disponibilidade para colaboração frequente e feedback contínuo.
    \item \textbf{Equipes Muito Grandes ou Terceirizadas:} A comunicação orgânica (face a face) falha e a coordenação requer processos estruturados mais rígidos.
    \item \textbf{Cultura Organizacional Rígida:} A empresa resiste à autonomia das equipes, prefere microgerenciamento e tem baixa tolerância a mudanças de escopo.
  \end{itemize}
\end{frame}
