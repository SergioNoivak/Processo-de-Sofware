\section{Modelo de Processo Unificado}

\begin{frame}{Modelo de Processo Unificado (PU)}
  \begin{itemize}
    \item \textbf{Fusão de Modelos:} Tentativa de aproveitar os melhores recursos dos modelos tradicionais de processo de software.
    \item \textbf{Princípios Ágeis:} Implementa muitos dos melhores princípios do desenvolvimento ágil de software.
    \item \textbf{Foco no Cliente:} Reconhece a importância da comunicação contínua e efetiva com o cliente.
    \item \textbf{Casos de Uso:} Utiliza métodos racionalizados (casos de uso) para descrever a visão do cliente sobre o sistema.
  \end{itemize}
\end{frame}

\begin{frame}{Modelo de Processo Unificado (PU)}
  \begin{itemize}
    \item \textbf{Papel da Arquitetura:} Enfatiza fortemente o papel central da arquitetura de software no desenvolvimento.
    \item \textbf{Metas Arquiteturais:} Ajuda o arquiteto a focar em:
      \begin{itemize}
        \item Compreensibilidade do sistema
        \item Confiança em mudanças futuras
        \item Reutilização de componentes
      \end{itemize}
    \item \textbf{Fluxo Iterativo e Incremental:} Sugere um fluxo de processo que proporciona uma evolução contínua.
    \item \textbf{Sensação Evolucionária:} Essencial no contexto do desenvolvimento de software moderno.
  \end{itemize}
\end{frame}

  \begin{columns}

      \begin{itemize}
        \item Descreve a interação entre o sistema e o usuário (ou outro sistema).

        \item Representa um resultado de valor para um \textbf{Ator}.

\begin{frame}{Fase de Concepção (PU)}
  \pumap{1}
  \begin{itemize}
    \item \textbf{Foco:} Comunicação com o cliente e planejamento.
    \item \textbf{Requisitos:} Descritos em um conjunto de casos de uso preliminares.
    \item \textbf{Arquitetura:} Define recursos e funções esperados por cada categoria de usuário.
    \item \textbf{Planejamento:} Identifica recursos, avalia os principais riscos e define um cronograma preliminar para os incrementos.
  \end{itemize}
\end{frame}

\begin{frame}{Fase de Elaboração (PU)}
  \pumap{2}
  \begin{itemize}
    \item \textbf{Foco:} Atividades de planejamento e modelagem.
    \item \textbf{Ações:} Refina e expande os casos de uso preliminares.
    \item \textbf{Arquitetura Base:} Criação considerando cinco visões: modelo de caso de uso, análise, projeto, implementação e entrega.
    \item \textbf{Ajustes:} Normalmente, as modificações no planejamento preliminar são feitas nesta fase.
  \end{itemize}
\end{frame}

\begin{frame}{Fase de Construção (PU)}
  \pumap{3}
  \begin{itemize}
    \item \textbf{Foco:} Idêntica à atividade de construção do processo genérico.
    \item \textbf{Implementação:} Código-fonte para todos os recursos do incremento de software.
    \item \textbf{Testes e Integração:} Testes de unidade para componentes, montagem e testes de integração.
    \item \textbf{Aceitação:} Casos de uso formam pacotes de testes executados antes da próxima fase.
  \end{itemize}
\end{frame}

\begin{frame}{Fase de Transição (PU)}
  \pumap{4}
  \begin{itemize}
    \item \textbf{Foco:} Últimos estágios de construção e primeira parte de emprego genérico (entrega e feedback).
    \item \textbf{Testes Beta:} Software entregue aos usuários para avaliação.
    \item \textbf{Feedback:} Relata defeitos e mudanças necessárias.
    \item \textbf{Conclusão:} O incremento torna-se uma versão utilizável do software.
  \end{itemize}
\end{frame}

\begin{frame}{Fase de Produção e Adaptação (PU)}
  \pumap{5}
  \begin{itemize}
    \item \textbf{Produção:} Coincide com a atividade de entrega genérica.
      \begin{itemize}
        \item Monitora o uso contínuo, provê suporte, gere relatórios de defeitos e mudanças.
      \end{itemize}
    \item \textbf{Concomitância:} Fases podem ocorrer de forma simultânea e escalonada.
    \item \textbf{Adaptação:} A equipe adapta o processo (ações, tarefas, artefatos) às suas necessidades; nem toda tarefa é conduzida em todos os projetos.
  \end{itemize}
\end{frame}

\section{Modelo Cascata}

\begin{frame}{Modelo Cascata: Quando utilizar?}
  \begin{itemize}
    \item \textbf{Requisitos Compreendidos:} Quando os requisitos de um problema são bem definidos e razoavelmente estáveis.
    \item \textbf{Fluxo Linear:} O trabalho flui da comunicação à entrega de modo linear.
    \item \textbf{Aperfeiçoamentos:} Comum em adaptações bem-definidas de sistemas existentes (ex.: adaptação em software contábil devido a novas normas).
    \item \textbf{Projetos Novos:} Pode ocorrer em novos esforços de desenvolvimento, de forma restrita.
  \end{itemize}
\end{frame}

\begin{frame}{Modelo Cascata: O que é?}
  \begin{itemize}
    \item Também chamado de \textbf{modelo sequencial linear}.
    \item Sugere uma abordagem sequencial e sistemática para o desenvolvimento de software.
    \item \textbf{Fases principais:}
      \begin{enumerate}
        \item Especificação dos requisitos do cliente.
        \item Planejamento.
        \item Modelagem.
        \item Construção.
        \item Entrega e Suporte contínuo do software concluído.
      \end{enumerate}
  \end{itemize}
\end{frame}

\begin{frame}{Modelo Cascata: Histórico e Críticas}
  \begin{itemize}
    \item \textbf{Histórico:} É o paradigma mais antigo da engenharia de software (mais de cinco décadas).
    \item \textbf{Eficácia Questionada:} Ao longo do tempo, as críticas a esse modelo de processo cresceram.
    \item \textbf{Mudança de Cenário:} Devido às características dinâmicas dos projetos modernos, até mesmo seus defensores mais árduos questionam sua eficácia nos dias atuais.
  \end{itemize}
\end{frame}

\section{Comparação entre Modelos}

\begin{frame}{Comparação entre Modelos de Processo (Parte 1)}
  \vspace{-0.2cm}
  {\scriptsize
  \begin{tabular}{|p{1.8cm}|p{4cm}|p{4cm}|}
    \hline
    \textbf{Modelo} & \textbf{Prós} & \textbf{Contras} \\
    \hline
    \textbf{Cascata} &
    \begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}
      \item Fácil de entender e planejar
      \item Funciona para projetos pequenos e bem compreendidos
      \item Análise e teste simples e diretos
    \end{itemize} &
    \begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}
      \item Não se adapta bem a mudanças
      \item Teste ocorre nas fases finais
      \item Aprovação do cliente vem no final
    \end{itemize} \\
    \hline
    \textbf{Prototipação} &
    \begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}
      \item Impacto de alterações nos requisitos é reduzido
      \item Cliente se envolve desde o início
      \item Funciona bem para projetos pequenos
      \item Probabilidade de rejeição reduzida
    \end{itemize} &
    \begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}
      \item Envolvimento do cliente pode causar atrasos
      \item Tentação de ``embalar'' o protótipo

\section{Dualidade Produto e Processo}

\begin{frame}{Processo Fraco vs. Confiança Excessiva}
  \begin{columns}
    \column{0.5\textwidth}
      \begin{itemize}
        \item Se o \textbf{processo for fraco}, o produto final sofrerá consequências.
        \item Uma \textbf{confiança excessiva e obsessiva} no processo é igualmente perigosa.
        \item Margaret Davis [Dav95a]: a cada $\sim$10 anos, a comunidade de software redefine ``o problema'', mudando o foco de itens do \textit{produto} para itens do \textit{processo}.
      \end{itemize}

\begin{frame}{A Dualidade Produto e Processo}
  \begin{columns}
    \column{0.5\textwidth}
      \begin{itemize}
        \item As oscilações entre foco em produto e processo \textbf{não resolvem ``o problema''}.
        \item Produto e processo devem ser tratados como uma \textbf{dualidade} (coexistência), não como uma \textbf{dicotomia} (oposição).
        \item Não se pode compreender o artefato completo olhando \textit{apenas} como processo ou \textit{apenas} como produto.
        \item A reutilização de produtos aumenta a satisfação e reforça a urgência de aceitar essa dualidade.
      \end{itemize}
    \column{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.85, transform shape]
        % Círculos sobrepostos (dualidade)
        \draw[thick, fill=blue!15, fill opacity=0.7] (-0.6, 0) circle (1.3);

\end{frame}

\begin{frame}{Satisfação Criativa no Software}
  \begin{itemize}
    \item As pessoas obtêm satisfação tanto do \textbf{processo criativo} quanto do \textbf{produto final}.
    \item \textbf{Analogias:}
      \begin{itemize}
        \item Um \textit{artista} sente prazer com suas pinceladas e com o quadro finalizado.
        \item Um \textit{escritor} sente prazer com a busca da metáfora e com o livro concluído.
      \end{itemize}
    \item Como \textbf{profissional de software criativo}, deve-se extrair satisfação do processo \textit{e} do produto final.
    \item A \textbf{dualidade produto e processo} é um elemento importante para manter pessoas criativas engajadas à medida que a engenharia de software evolui.
  \end{itemize}
\end{frame}
